    term: 'Job Plan'
    slug: 'job-plan'

    term: 'Job Plan Materials / Labor / Services / Tools'
    slug: 'job-plan-materials-labor-services-tools'

    term: 'Plans Tab (Planned Work)'
    slug: 'plans-tab-planned-work'

    term: 'Reservation (Reserved Items)'
    slug: 'reservation-reserved-items'
    definition: 'Holding stock aside for a specific work order or request so it is not used by someone else. Reservations can be hard or soft depending on settings.'
  - step: 7
    term: 'Work Order Estimate / Estimated Cost'
    slug: 'work-order-estimate-estimated-cost'
